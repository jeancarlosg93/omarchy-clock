cmake_minimum_required(VERSION 3.22)
project(OmarchyClock VERSION 2.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 DBus Sql Widgets Network)
add_library(timekeeper STATIC timekeeper.cpp timekeeper.h)
target_link_libraries(timekeeper PUBLIC Qt6::Gui Qt6::Sql)
qt_add_resources(timekeeper zoneassets PREFIX "/" FILES data/timezone-cities.json)
file(GLOB UI_FILES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS "ui/*.qml")
list(APPEND UI_FILES ui/WorldLand.js)
qt_add_executable(omarchy-clock main.cpp desktop.cpp desktop.h clockmodel.cpp clockmodel.h)
# The shell plugin is deployed separately; only dynamic Qt imports belong to this app.
set_target_properties(omarchy-clock PROPERTIES QT_QML_MODULE_NO_IMPORT_SCAN TRUE)
qt_add_resources(omarchy-clock assets PREFIX "/" FILES Main.qml omarchy-clock.svg ${UI_FILES})
target_link_libraries(omarchy-clock PRIVATE timekeeper Qt6::Quick Qt6::QuickControls2 Qt6::DBus Qt6::Widgets Qt6::Network)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(clock-tests tests.cpp clockmodel.cpp clockmodel.h)
  target_link_libraries(clock-tests PRIVATE Qt6::Gui Qt6::DBus)
  add_test(NAME timing COMMAND clock-tests)
  qt_add_executable(suite-tests suite-tests.cpp)
  target_link_libraries(suite-tests PRIVATE timekeeper)
  add_test(NAME suite COMMAND suite-tests)
  qt_add_executable(ui-tests ui-tests.cpp desktop.cpp desktop.h clockmodel.cpp clockmodel.h)
  set_target_properties(ui-tests PROPERTIES QT_QML_MODULE_NO_IMPORT_SCAN TRUE)
  qt_add_resources(ui-tests assets PREFIX "/" FILES Main.qml omarchy-clock.svg ${UI_FILES})
  target_link_libraries(ui-tests PRIVATE timekeeper Qt6::Quick Qt6::QuickControls2 Qt6::DBus Qt6::Test Qt6::Widgets Qt6::Network)
  add_test(NAME interface COMMAND ui-tests)
  set_tests_properties(interface PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_QUICK_BACKEND=software")
endif()
install(TARGETS omarchy-clock RUNTIME DESTINATION bin)
install(FILES omarchy-clock.desktop DESTINATION share/applications)
install(FILES omarchy-clock.svg DESTINATION share/icons/hicolor/scalable/apps)
